"use client";

import { useState, useEffect } from "react";
import { useQuery, useMutation } from "@tanstack/react-query";
import { useGetIdentity } from "@refinedev/core";
import { UserIdentity } from "@components/auth/interfaces";
import { axiosInstance } from "@providers/auth-provider/axiosInstance";
import { getRoleBooleans } from "lib/auth/roles";
// import DashboardLayout from "@components/shared/DashboardLayout";
import { ADMIN_DASHBOARD_CONFIG } from "@components/dashboard/admin/adminConfig";
// import AdminWorkArea from "@components/dashboard/admin/AdminWorkArea";
// import { LOCAL_STORAGE_KEYS } from "lib/utils/localStorage";

interface ToDoItem {
  id: number;
  title: string;
  is_completed: boolean;
  completed_date: string | null;
}

/**
 * ADMIN DASHBOARD
 *
 * Site-wide administration dashboard for system operators only.
 *
 * This dashboard is ONLY for people who run the entire site.
 * Group administration and content management should be accessed
 * through the member dashboard's expanded functionality.
 *
 * Features:
 * - System health monitoring
 * - Platform-wide user metrics
 * - Revenue and subscription management
 * - Site-wide content moderation
 * - Critical system settings
 *
 * ARCHITECTURE FOLLOWS MEMBER DASHBOARD PATTERN:
 * - Early auth checks to prevent unnecessary API calls
 * - Dynamic page titles
 * - Consistent component structure
 * - No auto-refresh intervals (manual refresh only)
 */
export default function AdminDashboard() {
  const { data: identity, isLoading: identityLoading } = useGetIdentity<UserIdentity>();

  // Set dynamic title
  useEffect(() => {
    if (identity) {
      document.title = "System Admin - Mixtape Crossroads";
    }
  }, [identity]);

  // EARLY RETURNS - Prevent unnecessary API calls
  if (identityLoading) {
    return <div>Loading admin dashboard...</div>;
  }

  if (!identity) {
    return <div>Authentication required...</div>;
  }

  // Verify admin permissions
  const { isAdmin } = getRoleBooleans(identity);
  if (!isAdmin) {
    return <div>Admin access required.</div>;
  }

  // NOW safe to make API calls
  const { data: systemStats = null } = useQuery({
    queryKey: ["admin-system-stats"],
    queryFn: async () => {
      const res = await axiosInstance.get("/api/admin/system-stats");
      return res.data;
    },
    enabled: !!identity?.id,
    // NO refetchInterval - manual refresh only
  });

  // Fetch todos ONLY when needed (no auto-refresh)
  const { data: todos = [], refetch: refetchTodos } = useQuery<ToDoItem[]>({
    queryKey: ["admin-todos"],
    queryFn: async () => {
      const res = await axiosInstance.get("/api/content/todo");
      return res.data as ToDoItem[];
    },
    enabled: !!identity?.id,
    // REMOVED: refetchInterval and onSuccess
  });

  // Fetch system-wide user metrics
  const { data: userMetrics = null } = useQuery({
    queryKey: ["admin-user-metrics"],
    queryFn: async () => {
      const res = await axiosInstance.get("/api/admin/user-metrics");
      return res.data;
    },
    enabled: !!identity?.id,
  });

  // Complete todo mutation (simplified)
  const completeTodo = useMutation({
    mutationFn: (id: number) => {
      return axiosInstance.patch(`/api/content/todo/${id}`, {
        is_completed: true,
        completed_date: new Date().toISOString(),
      });
    },
    onSuccess: () => {
      refetchTodos(); // Manual refetch after action
    },
  });

  // Work area wrapper
  const WorkAreaWrapper = (props: any) => {
    return (
      <AdminWorkArea
        {...props}
        identity={identity}
        todos={todos}
        onCompleteTodo={(id: number) => completeTodo.mutate(id)}
        systemStats={systemStats}
        userMetrics={userMetrics}
        onRefreshTodos={() => refetchTodos()}
      />
    );
  };

  return (
    <DashboardLayout
      title="System Administration"
      menuItems={ADMIN_DASHBOARD_CONFIG.menuItems}
      defaultSection={ADMIN_DASHBOARD_CONFIG.defaultSection}
      localStorageKey={ADMIN_DASHBOARD_CONFIG.localStorageKey}
      WorkAreaComponent={WorkAreaWrapper}
      workAreaProps={{}}
      loading={identityLoading}
    />
  );
}
