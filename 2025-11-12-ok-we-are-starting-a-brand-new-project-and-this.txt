  Code Review: Authentication Architecture

  Overview

  Your authentication setup uses Refine's auth provider pattern with JWT tokens (access + refresh) and httpOnly cookies. The architecture is well-structured with clear separation of concerns.

  ---
  Security Analysis

  ✅ Good Security Practices

  1. httpOnly Refresh Tokens
    - withCredentials: true in axios config (line 13 of axiosInstance.ts)
    - Refresh tokens stored as httpOnly cookies (protected from XSS)
  2. Token Expiry Validation
    - 30-second buffer before expiry (utils.ts:30)
    - Prevents race conditions
  3. Automatic Token Refresh
    - Axios interceptor handles 401 responses (axiosInstance.ts:44-73)
    - Prevents unnecessary logouts
  4. Authorization Header Management
    - Request interceptor adds Bearer token (axiosInstance.ts:17-28)
    - Properly updates on refresh

  🚨 Security Issues & Recommendations

  CRITICAL: Access Token in localStorage

  Problem:
  // authProvider.ts:123-124
  safeSetLocalStorage("access_token", response.data.access);
  safeSetLocalStorage("access_token_expiry", String(expiresAt));

  Risk: Access tokens in localStorage are vulnerable to XSS attacks. If an attacker injects JavaScript, they can steal the token.

  Recommendation:

  1. Create Auth Context

  // src/lib/auth/AuthContext.tsx
  'use client';

  import { createContext, useContext, useState, useEffect, ReactNode } from 'react';
  import { UserIdentity } from '@/types/auth';
  import { getAccessToken, clearAccessToken } from '@/lib/auth/tokenStorage';
  import { fetchAndValidateUser, performLogout } from '@/lib/auth/api';

  interface AuthContextType {
    user: UserIdentity | null;
    isLoading: boolean;
    isAuthenticated: boolean;
    login: (identifier: string, password: string) => Promise<void>;
    logout: () => Promise<void>;
    refreshUser: () => Promise<void>;
  }

  const AuthContext = createContext<AuthContextType | undefined>(undefined);

  export function AuthProvider({ children }: { children: ReactNode }) {
    const [user, setUser] = useState<UserIdentity | null>(null);
    const [isLoading, setIsLoading] = useState(true);

    // Check auth on mount
    useEffect(() => {
      checkAuth();
    }, []);

    const checkAuth = async () => {
      try {
        const token = getAccessToken();

        if (!token) {
          setUser(null);
          setIsLoading(false);
          return;
        }

        const userData = await fetchAndValidateUser(token);
        setUser(userData);
      } catch (error) {
        console.error('Auth check failed:', error);
        setUser(null);
      } finally {
        setIsLoading(false);
      }
    };

    const login = async (identifier: string, password: string) => {
      const response = await fetch(`${process.env.NEXT_PUBLIC_ROOT_API_URL}/api/auth/token`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        credentials: 'include',
        body: JSON.stringify({ identifier, password }),
      });

      const data = await response.json();

      if (!data.success) {
        throw new Error(data.detail || 'Login failed');
      }

      // Store token in memory
      const expiresAt = data.access_expires * 1000;
      setAccessToken(data.access, expiresAt);

      // Fetch user data
      const userData = await fetchAndValidateUser(data.access);
      setUser(userData);
    };

    const logout = async () => {
      await performLogout();
      clearAccessToken();
      setUser(null);
    };

    const refreshUser = async () => {
      await checkAuth();
    };

    return (
      <AuthContext.Provider
        value={{
          user,
          isLoading,
          isAuthenticated: !!user,
          login,
          logout,
          refreshUser,
        }}
      >
        {children}
      </AuthContext.Provider>
    );
  }

  export function useAuth() {
    const context = useContext(AuthContext);
    if (context === undefined) {
      throw new Error('useAuth must be used within AuthProvider');
    }
    return context;
  }

  export function middleware(request: NextRequest) {
    // Check for refresh token cookie
    const refreshToken = request.cookies.get('refresh_token');

    const isAuthPage = request.nextUrl.pathname.startsWith('/login') ||
                       request.nextUrl.pathname.startsWith('/signup');

    const isProtectedPage = request.nextUrl.pathname.startsWith('/dashboard') ||

    // Redirect to login if accessing protected page without token
    if (isProtectedPage && !refreshToken) {

    // Redirect to dashboard if accessing auth page with valid token
    if (isAuthPage && refreshToken) {
      return NextResponse.redirect(new URL('/dashboard', request.url));

  // src/lib/auth/permissions.ts

  export type Role = 'admin' | 'steward' | 'member';

  export type Permission =
    | 'users:read'
    | 'users:write'
    | 'posts:read'
    | 'posts:write'
    | 'posts:delete';

  const rolePermissions: Record<Role, Permission[]> = {
    admin: [
      'users:read',
      'users:write',
      'posts:read',
      'posts:write',
      'posts:delete',
    ],
    steward: [
      'users:read',
      'posts:read',
      'posts:write',
    ],
    member: [
      'posts:read',
    ],
  };

  export function hasPermission(userRoles: Role[], permission: Permission): boolean {
    return userRoles.some(role =>
      rolePermissions[role]?.includes(permission)
    );
  }

  export function hasAnyPermission(userRoles: Role[], permissions: Permission[]): boolean {
    return permissions.some(perm => hasPermission(userRoles, perm));
  }

  export function hasAllPermissions(userRoles: Role[], permissions: Permission[]): boolean {
    return permissions.every(perm => hasPermission(userRoles, perm));
  }

  2. Permission Hook

  // src/lib/auth/usePermissions.ts
  import { useAuth } from './AuthContext';
  import { hasPermission, Permission, Role } from './permissions';

  export function usePermissions() {
    const { user } = useAuth();

    const can = (permission: Permission): boolean => {
      if (!user) return false;
      return hasPermission(user.roles as Role[], permission);
    };

    const isAdmin = user?.is_superuser || user?.is_staff || false;
    const isSteward = user?.roles?.includes('steward') || false;
    const isMember = user?.roles?.includes('member') || false;

    return {
      can,
      isAdmin,
      isSteward,
      isMember,
    };
  }
