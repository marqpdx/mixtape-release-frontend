"use client";

import {
  Box,
  Heading,
  Text,
  Button,
  Container,
  Flex,
  Badge,
  HStack,
  VStack,
  Grid,
  GridItem,
  AvatarGroup,
  Card,
  Link,
  SimpleGrid,
  Avatar,
} from "@chakra-ui/react";
import { Tabs } from "@chakra-ui/react";
import { useState } from "react";
import {
  IconUsers,
  IconFiles,
  IconSchool,
  IconInfoHexagon,
  IconCalendar,
  IconPhoto,
  IconMapPin,
  IconExternalLink,
  IconWorld,
  IconLock,
  IconEye
} from "@tabler/icons-react";
import { useColorModeValue } from "@components/ui/color-mode";
import { useNoticeboard } from "./writing/useNoticeboard";
import { GroupNoticeboard } from "./writing/GroupNoticeboard";

interface GroupLandingProps {
  group: any;
  userRole?: 'admin' | 'steward' | 'member' | null;
  onJoinGroup?: () => void;
  loading?: boolean;
}

// Visibility icons mapping
const VISIBILITY_ICONS = {
  public: IconWorld,
  private: IconLock,
  unlisted: IconEye,
};

export function GroupLanding({ group, userRole, onJoinGroup, loading = false }: GroupLandingProps) {
  const bgColor = useColorModeValue('white', 'gray.800');
  const borderColor = useColorModeValue('gray.200', 'gray.600');
  const sidebarBg = useColorModeValue('gray.50', 'gray.700');

  const isMember = userRole !== null;
  const canJoin = group.visibility === 'public' && group.join_policy !== 'closed';
  const [showFullDescription, setShowFullDescription] = useState(false);

  const { items: noticeboard, loading: nbLoading, error: nbError, hasMore, loadMore, refresh } =
    useNoticeboard(group.slug);

  const publicTabs = [
    { key: 'landing', label: 'Landing Page', icon: IconInfoHexagon },
    { key: 'joining', label: 'How to Join', icon: IconUsers },
  ];

  const memberTabs = [
    { key: 'overview', label: 'Overview', icon: IconInfoHexagon },
    { key: 'files', label: 'Files', icon: IconFiles },
    { key: 'courses', label: 'Courses', icon: IconSchool },
    { key: 'members', label: 'Members', icon: IconUsers },
  ];

  const tabsToShow = isMember ? memberTabs : publicTabs;

  const description = group.description || "No description provided.";
  const shouldTruncate = description.length > 300;
  const displayDescription = shouldTruncate && !showFullDescription
    ? description.substring(0, 300) + "..."
    : description;

  const VisibilityIcon = VISIBILITY_ICONS[group.visibility as keyof typeof VISIBILITY_ICONS] || IconWorld;

  return (
    <Box className="group-landing">
      {/* Hero Section - Simplified, no title */}
      <Box
        position="relative"
        h="240px"
        bgImage={group.background_image ? `url(${group.background_image})` : 'linear-gradient(135deg, #667eea 0%, #764ba2 100%)'}
        bgSize="cover"
        bgPos="center"
        borderRadius="lg"
        mb={4}
      >
        <Box
          position="absolute"
          bottom={0}
          left={0}
          right={0}
          bg="linear-gradient(transparent, rgba(0,0,0,0.7))"
          p={6}
          borderBottomRadius="lg"
        >
          <HStack align="end" gap={4}>
            <AvatarGroup>
              <Avatar.Root size="xl" border="3px solid white">
                <Avatar.Image src={group.avatar} />
                <Avatar.Fallback>{group.title?.charAt(0) || 'G'}</Avatar.Fallback>
              </Avatar.Root>
            </AvatarGroup>
            <Box flex="1">
              <Text color="gray.200" fontSize="lg" mb={3}>
                {group.tagline || "Building community through shared purpose"}
              </Text>
              {!isMember && canJoin && (
                <HStack>
                  <Button
                    colorScheme="green"
                    size="lg"
                    onClick={onJoinGroup}
                  >
                    Join Group
                  </Button>
                  <Button
                    variant="outline"
                    size="lg"
                    color="white"
                    borderColor="white"
                    _hover={{ bg: 'whiteAlpha.200' }}
                  >
                    Follow
                  </Button>
                </HStack>
              )}
            </Box>
          </HStack>
        </Box>

        {/* Quick Facts Bar - Now with icon for visibility */}
        <Box
          position="absolute"
          bottom={4}
          right={4}
          bg="whiteAlpha.900"
          backdropFilter="blur(10px)"
          borderRadius="lg"
          p={3}
          border="1px solid"
          borderColor="whiteAlpha.300"
        >
          <HStack gap={3} wrap="wrap">
            {/* Visibility Icon */}
            <HStack gap={1}>
              <VisibilityIcon size={14} />
              <Text fontSize="xs" fontWeight="medium" textTransform="capitalize">
                {group.visibility}
              </Text>
            </HStack>

            {/* Location */}
            {group.location && (
              <HStack gap={1}>
                <IconMapPin size={14} />
                <Text fontSize="xs">{group.location}</Text>
              </HStack>
            )}

            {/* Member count */}
            <HStack gap={1}>
              <IconUsers size={14} />
              <Text fontSize="xs">{group.member_count || 0} members</Text>
            </HStack>
          </HStack>
        </Box>
      </Box>

      <Container maxW="6xl">
        {/* Tabbed Content */}
        <Tabs.Root defaultValue={tabsToShow[0].key} variant="enclosed">
          <Tabs.List mb={0}>
            {tabsToShow.map((tab) => {
              const IconComponent = tab.icon;
              return (
                <Tabs.Trigger key={tab.key} value={tab.key}>
                  <HStack>
                    <IconComponent size={16} />
                    <Text>{tab.label}</Text>
                  </HStack>
                </Tabs.Trigger>
              );
            })}
            <Tabs.Indicator />
          </Tabs.List>

          {/* Landing Page / Overview Tab */}
          <Tabs.Content value={isMember ? 'overview' : 'landing'}>
            <Grid templateColumns={{ base: '1fr', lg: '2fr 1fr' }} gap={4}>
              {/* Main Content */}
              <GridItem>
                <VStack align="stretch" gap={4}>
                  {/* About Section */}
                  <Card.Root>
                    <Card.Header>
                      <Heading size="md">About This Group</Heading>
                    </Card.Header>
                    <Card.Body>
                      <Text whiteSpace="pre-line" lineHeight="tall">
                        {displayDescription}
                      </Text>
                      {shouldTruncate && (
                        <Button
                          variant="ghost"
                          size="sm"
                          mt={2}
                          onClick={() => setShowFullDescription(!showFullDescription)}
                        >
                          {showFullDescription ? 'Show less' : 'Read more'}
                        </Button>
                      )}
                    </Card.Body>
                  </Card.Root>

                  {/* Public Posts */}
                  <GroupNoticeboard group={group} />

                  {/* Public Events */}
                  <Card.Root>
                    <Card.Header>
                      <Flex justify="space-between" align="center">
                        <Heading size="md">Upcoming Events</Heading>
                        <Button variant="ghost" size="sm">
                          <IconCalendar size={16} style={{ marginRight: '4px' }} />
                          View calendar
                        </Button>
                      </Flex>
                    </Card.Header>
                    <Card.Body>
                      <VStack align="stretch" gap={3}>
                        {[1, 2, 3].map((i) => (
                          <HStack key={i} p={3} borderWidth="1px" borderRadius="md">
                            <Box textAlign="center" minW="60px">
                              <Text fontSize="sm" fontWeight="bold">Dec</Text>
                              <Text fontSize="xl" fontWeight="bold">{15 + i}</Text>
                            </Box>
                            <VStack align="start" gap={0} flex="1">
                              <Text fontWeight="bold">Community Workshop {i}</Text>
                              <Text fontSize="sm" color="gray.600">2:00 PM - 4:00 PM</Text>
                              <Text fontSize="sm" color="gray.500">Online Event</Text>
                            </VStack>
                          </HStack>
                        ))}
                      </VStack>
                    </Card.Body>
                  </Card.Root>

                  {/* Public Photos */}
                  <Card.Root>
                    <Card.Header>
                      <Flex justify="space-between" align="center">
                        <Heading size="md">Group Photos</Heading>
                        <Button variant="ghost" size="sm">
                          <IconPhoto size={16} style={{ marginRight: '4px' }} />
                          Open gallery
                        </Button>
                      </Flex>
                    </Card.Header>
                    <Card.Body>
                      <SimpleGrid columns={3} gap={2}>
                        {[1, 2, 3, 4, 5, 6].map((i) => (
                          <Box
                            key={i}
                            h="100px"
                            bg="gray.200"
                            borderRadius="md"
                            display="flex"
                            alignItems="center"
                            justifyContent="center"
                          >
                            <IconPhoto size={24} color="gray.500" />
                          </Box>
                        ))}
                      </SimpleGrid>
                    </Card.Body>
                  </Card.Root>
                </VStack>
              </GridItem>

              {/* Sidebar */}
              <GridItem>
                <VStack align="stretch" gap={4}>
                  {/* Group Stewards */}
                  <Card.Root bg={sidebarBg}>
                    <Card.Header>
                      <Heading size="sm">Group Stewards</Heading>
                    </Card.Header>
                    <Card.Body>
                      <VStack align="stretch" gap={3}>
                        {[1, 2].map((i) => (
                          <HStack key={i}>
                            <AvatarGroup>
                              <Avatar.Root size="sm">
                                <Avatar.Fallback>S{i}</Avatar.Fallback>
                              </Avatar.Root>
                            </AvatarGroup>
                            <VStack align="start" gap={0}>
                              <Text fontSize="sm" fontWeight="bold">Steward Name</Text>
                              <Text fontSize="xs" color="gray.500">Group Admin</Text>
                            </VStack>
                          </HStack>
                        ))}
                      </VStack>
                    </Card.Body>
                  </Card.Root>

                  {/* Featured Course */}
                  <Card.Root bg={sidebarBg}>
                    <Card.Header>
                      <Heading size="sm">Featured Course</Heading>
                    </Card.Header>
                    <Card.Body>
                      <VStack align="start" gap={2}>
                        <Text fontWeight="bold">Permaculture Basics</Text>
                        <Text fontSize="sm" color="gray.600">
                          Learn the fundamentals of sustainable design
                        </Text>
                        <Button size="sm" variant="outline" w="full">
                          Learn More
                        </Button>
                      </VStack>
                    </Card.Body>
                  </Card.Root>

                  {/* Pinned Announcement */}
                  <Card.Root bg={sidebarBg}>
                    <Card.Header>
                      <Heading size="sm">Announcement</Heading>
                    </Card.Header>
                    <Card.Body>
                      <Text fontSize="sm">
                        Welcome to our community! Join us for our monthly meetup next Saturday.
                      </Text>
                    </Card.Body>
                  </Card.Root>

                  {/* Contact Links */}
                  <Card.Root bg={sidebarBg}>
                    <Card.Header>
                      <Heading size="sm">Connect</Heading>
                    </Card.Header>
                    <Card.Body>
                      <VStack align="stretch" gap={2}>
                        <Link href="#" fontSize="sm">
                          <HStack>
                            <IconExternalLink size={14} />
                            <Text>Website</Text>
                          </HStack>
                        </Link>
                        <Link href="#" fontSize="sm">
                          <HStack>
                            <IconExternalLink size={14} />
                            <Text>Discord</Text>
                          </HStack>
                        </Link>
                      </VStack>
                    </Card.Body>
                  </Card.Root>
                </VStack>
              </GridItem>
            </Grid>
          </Tabs.Content>

          {/* Other tab contents remain the same */}
          <Tabs.Content value="joining">
            <JoiningOptionsContent group={group} isMember={isMember} onJoinGroup={onJoinGroup} />
          </Tabs.Content>

          {isMember && (
            <>
              <Tabs.Content value="files">
                <FilesTabContent group={group} />
              </Tabs.Content>
              <Tabs.Content value="courses">
                <CoursesTabContent group={group} />
              </Tabs.Content>
              <Tabs.Content value="members">
                <MembersTabContent group={group} />
              </Tabs.Content>
            </>
          )}
        </Tabs.Root>
      </Container>
    </Box>
  );
}

// Tab content components remain the same
function JoiningOptionsContent({ group, isMember, onJoinGroup }: any) {
  return (
    <Card.Root maxW="2xl" mx="auto">
      <Card.Header>
        <Heading size="md">
          {isMember ? "Invite Others" : "Join This Group"}
        </Heading>
      </Card.Header>
      <Card.Body>
        {isMember ? (
          <VStack gap={4}>
            <Text>Share this group with others who might be interested:</Text>
            <Button colorScheme="green" w="full">Share Group</Button>
          </VStack>
        ) : (
          <VStack gap={4}>
            <Text>
              {group.join_policy === 'open' && "This group is open for anyone to join!"}
              {group.join_policy === 'approval' && "Membership requires approval from group admins."}
              {group.join_policy === 'invite' && "This group is invite-only."}
            </Text>
            {group.join_policy !== 'closed' && onJoinGroup && (
              <Button colorScheme="green" size="lg" w="full" onClick={onJoinGroup}>
                {group.join_policy === 'approval' ? "Request to Join" : "Join Group"}
              </Button>
            )}
            {group.joining_instructions && (
              <Box p={4} bg="gray.50" borderRadius="md" w="full">
                <Text fontSize="sm" whiteSpace="pre-line">
                  {group.joining_instructions}
                </Text>
              </Box>
            )}
          </VStack>
        )}
      </Card.Body>
    </Card.Root>
  );
}
