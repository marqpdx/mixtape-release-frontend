"use client";

import { useForm } from "@refinedev/react-hook-form";
import {
  Box,
  Button,
  RadioGroup,
  Stack,
  Text,
  Card,
  Heading,
} from "@chakra-ui/react";
import { createStandaloneToast } from "@chakra-ui/toast";
import { axiosInstance } from "@providers/auth-provider/axiosInstance";
import { useEffect, useRef, useState, memo, useCallback } from "react";
import { DatePickerInput } from "@components/forms/DatePickerField";
import { Input } from "@theme/recipes/input.recipe";
import GroupVisibilitySelect from "@components/groups/GroupVisibilitySelect";

import { useWatch } from "react-hook-form";

const { toast } = createStandaloneToast();

const groupTypeOptions = [
  { id: "community", label: "Community", value: "community" },
  { id: "circle", label: "Circle", value: "circle" },
];

interface GroupCreateFormProps {
  onSuccess?: (slug: string) => void;
  onSuccessAndEdit?: (slug: string) => void;
  showCard?: boolean;
  title?: string;
}

// DEBUG: Track render count
let renderCount = 0;

const GroupCreateForm = memo(function GroupCreateForm({
  onSuccess,
  onSuccessAndEdit,
  showCard = true,
  title = "Create New Group"
}: GroupCreateFormProps) {

  // DEBUG: Log every render with details
  renderCount++;
  console.log(`🔍 DEBUG: GroupCreateForm render #${renderCount}`, {
    onSuccess: onSuccess?.toString().slice(0, 50),
    onSuccessAndEdit: onSuccessAndEdit?.toString().slice(0, 50),
    showCard,
    title
  });

  // DEBUG: Track prop changes
  const prevProps = useRef({ onSuccess, onSuccessAndEdit, showCard, title });
  useEffect(() => {
    const prev = prevProps.current;
    if (prev.onSuccess !== onSuccess) console.log('🔍 DEBUG: onSuccess prop changed');
    if (prev.onSuccessAndEdit !== onSuccessAndEdit) console.log('🔍 DEBUG: onSuccessAndEdit prop changed');
    if (prev.showCard !== showCard) console.log('🔍 DEBUG: showCard prop changed');
    if (prev.title !== title) console.log('🔍 DEBUG: title prop changed');
    prevProps.current = { onSuccess, onSuccessAndEdit, showCard, title };
  });

  const {
    register,
    handleSubmit,
    setValue,
    control,
    formState: { isSubmitting },
  } = useForm({
    mode: "onBlur",
    defaultValues: {
      title: "",
      description: "",
      visibility: "public",
      group_type: "community",
      tagline: "",
      start_date: null,
      end_date: null,
    }
  });

  // DEBUG: Track form state changes
  const visibility = useWatch({ control, name: "visibility" });
  const groupType = useWatch({ control, name: "group_type" });
  const startDate = useWatch({ control, name: "start_date" });
  const titleValue = useWatch({ control, name: "title" });

  // DEBUG: Log when watched values change
  useEffect(() => {
    console.log('🔍 DEBUG: Form state changed', { visibility, groupType, titleValue });
  }, [visibility, groupType, titleValue]);

  // DEBUG: Focus tracking
  const titleInputRef = useRef<HTMLInputElement>(null);

  useEffect(() => {
    console.log('🔍 DEBUG: Component mounted, setting up focus');

    const focusTitle = () => {
      const titleField = document.querySelector('[name="title"]') as HTMLInputElement;
      console.log('🔍 DEBUG: Attempting to focus title field:', titleField);

      if (titleField) {
        titleField.focus();
        console.log('🔍 DEBUG: Focus applied, active element:', document.activeElement);
      }
    };

    // Try multiple focus attempts
    focusTitle();
    const timer1 = setTimeout(focusTitle, 50);
    const timer2 = setTimeout(focusTitle, 200);

    return () => {
      clearTimeout(timer1);
      clearTimeout(timer2);
    };
  }, []); // Only on mount

  // DEBUG: Track when title input loses focus
  const handleTitleFocus = () => {
    console.log('🔍 DEBUG: Title input focused');
  };

  const handleTitleBlur = () => {
    console.log('🔍 DEBUG: Title input blurred');
  };

  const createGroup = useCallback(async (values: any): Promise<string | null> => {
    try {
      const res = await axiosInstance.post("/api/groups", values);
      console.log('Group created:', res);
      toast({
        title: "Group Created",
        description: `Your ${values.group_type} was created successfully.`,
        status: "success",
        duration: 5000,
        isClosable: true,
      });
      return res.data?.slug ?? null;
    } catch (err) {
      console.error("Error creating group:", err);
      toast({
        title: "Error",
        description: "Group creation failed. Please try again.",
        status: "error",
        duration: 5000,
        isClosable: true,
      });
      return null;
    }
  }, []);

  const onSubmit = useCallback(async (values: any) => {
    console.log('Form values:', values);
    const slug = await createGroup(values);
    if (slug && onSuccess) {
      toast({
        title: "Success",
        description: "Group created successfully.",
        status: "success",
        duration: 4000,
        isClosable: true,
      });
      onSuccess(slug);
    }
  }, [createGroup, onSuccess]);

  const FormContent = () => (
    <Box as="form" onSubmit={handleSubmit(onSubmit)}>
      <Stack gap={6}>
        {showCard && (
          <Heading size="lg" color="green.500" mb={2}>
            {title}
          </Heading>
        )}

        <Stack gap={4}>
          <Box>
            <Text fontSize="sm" fontWeight="medium" mb={2} color="gray.600">
              Group Name *
            </Text>
            {/* DEBUG: Add focus/blur handlers and ref */}
            <Input
              ref={titleInputRef}
              {...register("title", {
                required: "Group name is required",
              })}
              placeholder="Enter group name"
              size="md"
              onFocus={handleTitleFocus}
              onBlur={handleTitleBlur}
              onChange={(e) => {
                console.log('🔍 DEBUG: Title input changed:', e.target.value);
                // Let react-hook-form handle the change
              }}
            />
          </Box>

          {/* Simplified form for debugging - remove other fields temporarily */}
          <Box>
            <Text fontSize="sm" fontWeight="medium" mb={2} color="gray.600">
              Description *
            </Text>
            <Input
              {...register("description", { required: "Description is required" })}
              placeholder="Describe your group's purpose"
              size="md"
            />
          </Box>
        </Stack>

        <Stack direction={{ base: "column", md: "row" }} gap={4} pt={4}>
          <Button
            type="submit"
            loading={isSubmitting}
            colorScheme="green"
            size="lg"
            flex={1}
          >
            Create Group
          </Button>
        </Stack>
      </Stack>
    </Box>
  );

  if (!showCard) {
    return <FormContent />;
  }

  return (
    <Box maxW="800px" mx="auto" mt={6}>
      <Card.Root>
        <Card.Body p={8}>
          <FormContent />
        </Card.Body>
      </Card.Root>
    </Box>
  );
}, (prevProps, nextProps) => {
  // DEBUG: Custom comparison function for memo
  const propsChanged = {
    onSuccess: prevProps.onSuccess !== nextProps.onSuccess,
    onSuccessAndEdit: prevProps.onSuccessAndEdit !== nextProps.onSuccessAndEdit,
    showCard: prevProps.showCard !== nextProps.showCard,
    title: prevProps.title !== nextProps.title,
  };

  const shouldRerender = Object.values(propsChanged).some(changed => changed);
  console.log('🔍 DEBUG: memo comparison', { propsChanged, shouldRerender });

  return !shouldRerender; // return true to skip re-render
});

export default GroupCreateForm;
