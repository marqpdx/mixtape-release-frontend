// src/components/groups/GroupsTable.tsx

"use client";

import {
  Box,
  Heading,
  Spinner,
  Text,
  HStack,
  VStack,
  Avatar,
  IconButton,
} from "@chakra-ui/react";
import { createColumnHelper, flexRender } from "@tanstack/react-table";
import { useTable } from "@refinedev/react-table";
import { Icons } from "@components/icons/IconMap";
import { useRouter } from "next/navigation";
import { IconEdit } from "@tabler/icons-react";
import { Group } from "./interfaces";
import { useColorModeValue } from "@components/ui/color-mode";

interface GroupQueryResponse {
  count: number;
  next: string | null;
  previous: string | null;
  results: Group[];
}

const columnHelper = createColumnHelper<Group>();

export default function GroupsTable() {
  const router = useRouter();
  const bgHover = useColorModeValue("gray.50", "gray.700");
  const borderColor = useColorModeValue("gray.200", "gray.600");

  const columns = [
    columnHelper.display({
      id: "group_info",
      header: "Groups",
      cell: ({ row }) => {
        const group = row.original;
        const GroupIcon = Icons.group[group.group_type];
        const isAdmin = true; // TODO: Replace with actual admin check

        const handleGroupClick = () => {
          router.push(`/groups/${group.slug}/show`);
        };

        const handleEditClick = (e: React.MouseEvent) => {
          e.stopPropagation();
          router.push(`/groups/${group.slug}/edit`);
        };

        return (
          <HStack
            gap={4}
            py={2}
            cursor="pointer"
            onClick={handleGroupClick}
            _hover={{ bg: bgHover }}
            rounded="md"
            px={3}
            mx={-3}
            transition="all 0.2s"
          >
            {/* Profile Image Section */}
            <Box flexShrink={0}>
              <Avatar.Root size="lg" cursor="pointer">
                <Avatar.Image
                  src={group.profile_image}
                  alt={`${group.title} profile`}
                />
                <Avatar.Fallback bg="green.100" color="green.700">
                  {GroupIcon ? (
                    <GroupIcon size={24} />
                  ) : (
                    group.title.charAt(0).toUpperCase()
                  )}
                </Avatar.Fallback>
              </Avatar.Root>
            </Box>

            {/* Content Section */}
            <VStack align="start" gap={1} flex={1} minW={0}>
              <HStack justify="space-between" w="full">
                <Text
                  fontWeight="semibold"
                  fontSize="md"
                  color="gray.900"
                  _dark={{ color: "white" }}
                  cursor="pointer"
                  _hover={{ color: "green.600" }}
                  transition="color 0.2s"
                  lineClamp={1}
                >
                  {group.title}
                </Text>

                <HStack gap={2} flexShrink={0}>
                  <Text
                    fontSize="xs"
                    color="gray.500"
                    textTransform="capitalize"
                    bg="gray.100"
                    _dark={{ bg: "gray.700" }}
                    px={2}
                    py={1}
                    rounded="full"
                  >
                    {group.group_type}
                  </Text>

                  {isAdmin && (
                    <IconButton
                      aria-label="Edit group"
                      size="sm"
                      variant="ghost"
                      colorScheme="green"
                      onClick={handleEditClick}
                      _hover={{ bg: "green.100" }}
                    >
                      <IconEdit size={16} />
                    </IconButton>
                  )}
                </HStack>
              </HStack>

              <Text
                fontSize="sm"
                color="gray.600"
                _dark={{ color: "gray.300" }}
                lineClamp={2}
                wordBreak="break-word"
              >
                {group.description || "No description available"}
              </Text>

              <Text fontSize="xs" color="gray.400" mt={1}>
                Created {new Date(group.created_at).toLocaleDateString()}
              </Text>
            </VStack>
          </HStack>
        );
      },
    }),
  ];

  const {
    getHeaderGroups,
    getRowModel,
    refineCore: { tableQuery },
  } = useTable({
    columns,
    refineCoreProps: {
      resource: "groups",
      // Add pagination and prevent continuous queries
      pagination: {
        pageSize: 10,
        current: 1,
      },
      // Disable real-time updates to prevent continuous queries
      liveMode: "off",
      // Add polling interval if you need periodic updates
      // pollingInterval: 30000, // 30 seconds, or remove entirely
    },
    initialState: {
      sorting: [{ id: "created_at", desc: true }],
    },
  });

  const { isLoading, data: queryData, error } = tableQuery;

  const groups: Group[] =
    ((queryData?.data as unknown) as GroupQueryResponse)?.results ?? [];

  if (isLoading) {
    return (
      <Box
        display="flex"
        justifyContent="center"
        alignItems="center"
        height="300px"
        flexDirection="column"
        gap={4}
      >
        <Spinner size="xl" color="green.500" />
        <Text color="gray.500">Loading groups...</Text>
      </Box>
    );
  }

  if (error) {
    return (
      <Box
        textAlign="center"
        py={10}
        border="1px solid"
        borderColor={borderColor}
        rounded="lg"
      >
        <Text color="red.500" fontSize="lg" fontWeight="medium">
          Failed to load groups
        </Text>
        <Text color="gray.500" mt={2}>
          Please try refreshing the page
        </Text>
      </Box>
    );
  }

  if (!groups.length) {
    return (
      <Box
        textAlign="center"
        py={16}
        border="1px solid"
        borderColor={borderColor}
        rounded="lg"
        bg={useColorModeValue("gray.50", "gray.800")}
      >
        <Text fontSize="lg" fontWeight="medium" color="gray.600" mb={2}>
          No groups found
        </Text>
        <Text color="gray.500">
          Create your first group to get started
        </Text>
      </Box>
    );
  }

  return (
    <Box>
      <Heading size="lg" mb={6} color="green.600">
        All Groups ({groups.length})
      </Heading>

      <Box
        border="1px solid"
        borderColor={borderColor}
        rounded="lg"
        overflow="hidden"
        bg={useColorModeValue("white", "gray.800")}
        shadow="sm"
      >
        {/* Header */}
        <Box
          px={6}
          py={4}
          bg={useColorModeValue("gray.50", "gray.700")}
          borderBottom="1px solid"
          borderColor={borderColor}
        >
          {getHeaderGroups().map((headerGroup) => (
            <Box key={headerGroup.id}>
              {headerGroup.headers.map((header) => (
                <Text
                  key={header.id}
                  fontWeight="semibold"
                  fontSize="sm"
                  color="gray.700"
                  _dark={{ color: "gray.200" }}
                  cursor={header.column.getCanSort() ? "pointer" : "default"}
                  onClick={header.column.getToggleSortingHandler?.()}
                  _hover={header.column.getCanSort() ? { color: "green.600" } : {}}
                >
                  {flexRender(header.column.columnDef.header, header.getContext())}
                  {header.column.getIsSorted() === "asc" && " ↑"}
                  {header.column.getIsSorted() === "desc" && " ↓"}
                </Text>
              ))}
            </Box>
          ))}
        </Box>

        {/* Body */}
        <Box>
          {getRowModel().rows.map((row, index) => (
            <Box
              key={row.id}
              px={6}
              py={3}
              borderBottom={
                index !== getRowModel().rows.length - 1
                  ? "1px solid"
                  : "none"
              }
              borderColor={borderColor}
              _last={{ borderBottom: "none" }}
            >
              {row.getVisibleCells().map((cell) => (
                <Box key={cell.id}>
                  {flexRender(cell.column.columnDef.cell, cell.getContext())}
                </Box>
              ))}
            </Box>
          ))}
        </Box>
      </Box>
    </Box>
  );
}
