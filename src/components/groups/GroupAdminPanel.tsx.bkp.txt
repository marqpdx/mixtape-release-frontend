import { useGetIdentity } from "@refinedev/core";
import { Button, Box, Heading, Stack } from "@chakra-ui/react";
import { useRouter } from "next/navigation";
import { UserIdentity } from "@components/auth/interfaces";
import { useEffect, useState } from "react";
import { axiosInstance } from "@providers/auth-provider/axiosInstance";

import { Group } from "./interfaces";
import GroupMembersTable from "./GroupMemberTable";

type GroupAdminPanelProps = {
  group: Group;
};

export const GroupAdminPanel = ({ group }: GroupAdminPanelProps) => {
  const router = useRouter();
  const [members, setMembers] = useState([]);
  const { data: identity, isLoading: identityLoading } = useGetIdentity<UserIdentity>();

  const isAdmin = identity?.id === group.created_at || identity?.is_staff;

  if (!isAdmin) return null;

  useEffect(() => {
    if (!group?.slug) return;

    axiosInstance
      .get(`/api/groups/${group.slug}/members`)
      .then((res) => {
        setMembers(res.data.results);
      });
  }, [group?.slug]);

  console.log('members', members)

  const handleSendMessage = (userId: number) => {
      // Replace this with actual logic
      console.log("Send message to", userId);
    };

  return (
    <Box mt={0} p={5} border="1px solid" borderColor="gray.200" borderRadius="md">
      <Heading size="sm" mb={3}>
        Admin Panel
      </Heading>
      <Stack direction="row" gap={4} p={4} border={'1px solid gray'} borderRadius={5}>
        <Button
          size="sm"
          variant="outline"
          onClick={() => router.push(`/groups/${group.slug}/edit`)}
        >
          Edit Group
        </Button>
        <Button
          size="sm"
          onClick={() => router.push(`/groups/${group.slug}/invite`)}
        >
          Invite New Members
        </Button>
      </Stack>
      <Box mt={5} p={4} border={'1px solid gray'} borderRadius={5}>
