// src/components/groups/MembersAndRoles.tsx

"use client";
console.log("component: MembersAndRoles is rendering");

import {
  Box,
  Heading,
  Button,
  HStack,
  AvatarGroup,
  Avatar,
  Table,
  useDisclosure,
  Text,
  chakra,
} from "@chakra-ui/react";
import Link from "next/link";
import { useEffect, useMemo, useState } from "react";
import AdminModal from "@components/admin/AdminModal";
import { GroupInviteForm } from "@components/groups/forms/GroupInviteForm";
import { axiosInstance } from "@providers/auth-provider/axiosInstance";
import { useInvitationStatusPoll } from "@hooks/useInvitationStatusPoll";

import { Member } from "./interfaces";
import { AnimatePresence, motion } from "framer-motion";

export interface MembersAndRolesProps {
  members: Member[];
  highlightedMemberId: number | null;
  setHighlightedMemberId: (id: number | null) => void;
  groupSlug: string;
}

export default function MembersAndRoles({
  members: initialMembers,
  highlightedMemberId,
  setHighlightedMemberId,
  groupSlug,
}: MembersAndRolesProps) {
  const [selectedMemberId, setSelectedMemberId] = useState<number | null>(null);
  // const [membersList, setMembersList] = useState<Member[]>(members);

  // const [members, setMembers] = useState<Member[]>(initialMembers); // rename incoming props to `props.members`
  const [members, setMembers] = useState<Member[]>(initialMembers ?? []);



  const [pollInvitationId, setPollInvitationId] = useState<number | null>(null);
  const { open, onOpen, onClose } = useDisclosure();

  const [inviteStatusMessage, setInviteStatusMessage] = useState<string | null>(null);
  const [inviteStatusType, setInviteStatusType] = useState<"info" | "success" | "error" | null>(null);

  const membersList = useMemo(() => {
    if (!Array.isArray(members)) return [];
    return members.map((m: any) => ({
      ...m,
      user: m.member_data,
    }));
  }, [members]);

  const refetchMembers = async () => {
    try {
      const res = await axiosInstance.get(`/api/groups/${groupSlug}/members`);
      const results = res.data?.results;
      if (Array.isArray(results)) {
        setMembers(results);
      } else {
        console.warn("refetchMembers got unexpected data:", res.data);
        setMembers([]); // fallback
      }
    } catch (err) {
      console.error("Failed to fetch members:", err);
      setMembers([]); // optional defensive fallback
    }
  };

  useEffect(() => {
    setSelectedMemberId(
      highlightedMemberId !== undefined ? highlightedMemberId : null
    );
  }, [highlightedMemberId]);

  const handleRowClick = (memberId: number) => {
    setSelectedMemberId(memberId);
    setHighlightedMemberId(memberId);
  };

  // ✅ Call the hook at the top level
  useInvitationStatusPoll({
    groupId: groupSlug,
    invitationId: pollInvitationId,
    onSuccess: () => {
      setInviteStatusMessage("✅ Invitation email sent successfully.");
      setInviteStatusType("success");
      setPollInvitationId(null);
    },
    onFailure: () => {
      setInviteStatusMessage("❌ Failed to send invitation email.");
      setInviteStatusType("error");
      setPollInvitationId(null);
    },
    onPending: () => {
      setInviteStatusMessage("ℹ️ Invitation email is still being sent. It might take a moment.");
      setInviteStatusType("info");
      setPollInvitationId(null);
    },
    maxAttempts: 3,
    delay: 4400,
  });

  useEffect(() => {
    if (inviteStatusType === "success") {
      const timer = setTimeout(() => {
        setInviteStatusMessage(null);
        setInviteStatusType(null);
      }, 3500);

      return () => clearTimeout(timer);
    }
  }, [inviteStatusType]);

  const MotionBox = chakra(motion.div);

  console.log("MembersAndRoles membersList", membersList);

  return (
    <>
      <Box>
        <HStack justify="space-between" mb={4} align="flex-start">
          <Heading size="md">Members & Roles</Heading>
          <HStack justify="space-between" mb={4} align="flex-start">
            <Box>
              <AnimatePresence>
                {inviteStatusMessage && (
                  <MotionBox
                    initial={{ opacity: 0 }}
                    animate={{ opacity: 1, transition: { duration: 0.3 } }}
                    exit={{ opacity: 0, transition: { duration: 0.3 } }}
                  >
                    <Text
                      mt={2}
                      fontSize="sm"
                      color={
                        inviteStatusType === "error"
                          ? "red.500"
                          : inviteStatusType === "success"
                          ? "green.600"
                          : "gray.600"
                      }
                    >
                      {inviteStatusMessage}
                    </Text>
                  </MotionBox>
                )}
              </AnimatePresence>
            </Box>
            <Button size="sm" onClick={onOpen}>
              Invite Member
            </Button>
          </HStack>
        </HStack>

        <Box overflowX="auto">
          <Table.Root>
            <Table.Header>
              <Table.Row>
                <Table.ColumnHeader>Avatar</Table.ColumnHeader>
                <Table.ColumnHeader>Name</Table.ColumnHeader>
                <Table.ColumnHeader>Role</Table.ColumnHeader>
                <Table.ColumnHeader>Joined</Table.ColumnHeader>
                <Table.ColumnHeader>Last Active</Table.ColumnHeader>
                <Table.ColumnHeader>Status</Table.ColumnHeader>
                <Table.ColumnHeader>Actions</Table.ColumnHeader>
              </Table.Row>
            </Table.Header>

            <Table.Body>
              {membersList.map((member) => (
                <Table.Row
                  key={member.id}
                  onClick={() => handleRowClick(member.id)}
                  bg={
                    member.id === highlightedMemberId
                      ? "blue.50"
                      : selectedMemberId === member.id
                      ? "gray.100"
                      : "transparent"
                  }
                  cursor="pointer"
                >
                  <Table.Cell>
                    <AvatarGroup size="sm">
                      <Avatar.Root>
                        {member.user.profile.avatar ? (
                          <Avatar.Image
                            src={member.user.profile.avatar}
                            alt={
                              member.user.profile.name ||
                              member.user.username
                            }
                          />
                        ) : (
                          <Avatar.Fallback>
                            {member.user.profile.name
                              ? member.user.profile.name.charAt(0)
                              : member.user.username.charAt(0)}
                          </Avatar.Fallback>
                        )}
                      </Avatar.Root>
                    </AvatarGroup>
                  </Table.Cell>

                  <Table.Cell>
                    <Link
                      href={`/member/profiles/${member.user.id}`}
                      color="blue.600"
                    >
                      {member.user.profile.name || member.user.username}
                    </Link>
                  </Table.Cell>

                  <Table.Cell>{member.role}</Table.Cell>
                  <Table.Cell>
                    {new Date(member.date_joined).toLocaleDateString()}
                  </Table.Cell>
                  <Table.Cell>
                    {member.last_active
                      ? new Date(member.last_active).toLocaleDateString()
                      : "—"}
                  </Table.Cell>
                  <Table.Cell>{member.status}</Table.Cell>
                  <Table.Cell>
                    <HStack gap={2}>
                      <Button size="xs" variant="outline">
                        Edit
                      </Button>
                      <Button size="xs" variant="outline" colorScheme="red">
                        Ban
                      </Button>
                      <Button size="xs" variant="outline">
                        Message
                      </Button>
                    </HStack>
                  </Table.Cell>
                </Table.Row>
              ))}
            </Table.Body>
          </Table.Root>
        </Box>
      </Box>

      {/* Modal for inviting a new member */}
      <AdminModal
        title="Invite New Member"
        isOpen={open}
        onClose={onClose}
      >
        <GroupInviteForm
          groupSlug={groupSlug}
          onSuccess={(invitationId: number) => {
            setInviteStatusMessage("⏳ Sending invitation email...");
            setInviteStatusType("info");

            onClose();
            refetchMembers();
            setPollInvitationId(invitationId);
          }}
        />
      </AdminModal>
    </>
  );
}
