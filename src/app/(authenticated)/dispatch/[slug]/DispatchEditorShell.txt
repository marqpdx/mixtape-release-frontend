"use client";

import { useEffect, useRef, useState, useMemo, useCallback } from "react";
import { Box, Heading, Spinner, Text, Flex, Button, Input } from "@chakra-ui/react";
import { axiosInstance } from "@providers/auth-provider/axiosInstance";
import TipTapCollabEditor from "@components/editor/TipTapCollabEditor";
import { useAuth } from "@/lib/auth/AuthContext";
import { useYjsSocketProvider } from "@/lib/dispatch/yjs/useYjsSocketProvider";
import { useCollabAutosave } from "@hooks/dispatch/useCollabAutosave";
import { IconUsers, IconDeviceFloppy } from "@tabler/icons-react";
import ShareCollaboratorsModal from "@components/dispatch/ShareCollaboratorsModal";

interface DispatchDocument {
  id: string;
  yjs_document_id: string;
  title: string;
  content: any;
}

export default function DispatchEditorShell({ slug }: { slug: string }) {
  const [doc, setDoc] = useState<DispatchDocument | null>(null);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [shareModalOpen, setShareModalOpen] = useState(false);
  const [title, setTitle] = useState<string>("");
  const titleSaveTimeoutRef = useRef<NodeJS.Timeout | null>(null);
  const editorRef = useRef<any>(null);

  const { user: identity, isLoading: identityLoading } = useAuth();

  // Memoize user object to prevent useYjsSocketProvider from recreating on every render
  const userInfo = useMemo(() => ({
    name: identity?.profile?.display_name || "Anonymous",
    color: "#3498db"
  }), [identity?.profile?.display_name]);

  // Memoize dispatch content object to prevent useYjsSocketProvider from recreating on every render
  const dispatchContentForYjs = useMemo(() => {
    if (!doc) return null;
    return { id: doc.id, yjs_document_id: slug };
  }, [doc?.id, slug]);

  // Initialize yjs socket provider for collaborative editing
  const { ydoc, provider, isReady } = useYjsSocketProvider(
    dispatchContentForYjs,
    {
      user: userInfo,
      enabled: !!identity && !identityLoading && !!doc,
      initialContent: doc?.content,
    }
  );

  // Collaborative autosave hook - saves both yjs_state and content_snapshot
  const { status: saveStatus, lastSaved, triggerSave } = useCollabAutosave({
    documentSlug: slug,
    ydoc,
    editor: editorRef.current,
    enabled: isReady,
    debounceMs: 2500,      // Wait 2.5s after last change
    maxWaitMs: 30000,      // Force save every 30s while typing
    onSaveStart: () => console.log('=¾ Collaborative save starting...'),
    onSaveSuccess: () => console.log(' Collaborative save succeeded'),
    onSaveError: (err) => console.error('L Collaborative save failed:', err),
  });

  // Fetch document on mount
  useEffect(() => {
    const fetchDoc = async () => {
      try {
        console.log("= Shell fetching document:", slug);
        const res = await axiosInstance.get(`/api/dispatch/content/${slug}`);
        console.log("=Ä Shell received document:", res.data);
        setDoc(res.data);
        setTitle(res.data.title || "");
        setError(null);
      } catch (err) {
        console.error("L Shell failed to load document:", err);
        setError("Failed to load document");
      } finally {
        setLoading(false);
      }
    };

    if (slug && !identityLoading) {
      fetchDoc();
    }
  }, [slug, identityLoading]);

  // Handle title changes with debounced autosave
  const handleTitleChange = (newTitle: string) => {
    setTitle(newTitle);

    if (titleSaveTimeoutRef.current) {
      clearTimeout(titleSaveTimeoutRef.current);
    }

    titleSaveTimeoutRef.current = setTimeout(async () => {
      try {
        console.log("=¾ Auto-saving title:", newTitle);
        await axiosInstance.patch(`/api/dispatch/content/${slug}`, {
          title: newTitle,
        });
        console.log(" Title saved");
      } catch (err) {
        console.error("L Failed to save title:", err);
      }
    }, 1000);
  };

  // Cleanup title save timeout on unmount
  useEffect(() => {
    return () => {
      if (titleSaveTimeoutRef.current) {
        clearTimeout(titleSaveTimeoutRef.current);
      }
    };
  }, []);

  if (loading) {
    return (
      <Box p={10} textAlign="center">
        <Spinner />
        <Text mt={2} fontSize="sm" color="gray.500">Loading document...</Text>
      </Box>
    );
  }

  if (error || !doc) {
    return (
      <Box p={10} textAlign="center">
        <Text color="red.500">{error || "Document not found."}</Text>
      </Box>
    );
  }

  if (!ydoc || !provider) {
    return (
      <Box p={10} textAlign="center">
        <Spinner />
        <Text mt={2} fontSize="sm" color="gray.500">
          Connecting to collaborative editing&
        </Text>
      </Box>
    );
  }

  return (
    <Box maxW="6xl" mx="auto" py={10} px={4}>
      {/* Header with title, save, and share buttons */}
      <Flex justify="space-between" align="center" mb={4} gap={4}>
        <Input
          value={title}
          onChange={(e) => handleTitleChange(e.target.value)}
          placeholder="Enter document title..."
          size="lg"
          fontSize="xl"
          fontWeight="semibold"
          flex="1"
          _focus={{
            borderColor: "blue.500",
            boxShadow: "0 0 0 1px var(--chakra-colors-blue-500)"
          }}
        />
        <Flex gap={2} flexShrink={0}>
          <Button
            variant="solid"
            size="sm"
            onClick={triggerSave}
            disabled={saveStatus === "saving"}
            colorPalette={saveStatus === "saved" ? "green" : saveStatus === "error" ? "red" : "blue"}
          >
            <IconDeviceFloppy size={18} style={{ marginRight: '6px' }} />
            {saveStatus === "saving" ? "Saving..." : saveStatus === "saved" ? "Saved" : saveStatus === "error" ? "Error" : "Save"}
          </Button>
          <Button
            variant="outline"
            size="sm"
            onClick={() => setShareModalOpen(true)}
          >
            <IconUsers size={18} style={{ marginRight: '6px' }} />
            Share
          </Button>
        </Flex>
      </Flex>

      {/* Save status indicator */}
      {lastSaved && (
        <Text fontSize="xs" color="gray.500" mb={2}>
          Last saved {lastSaved.toLocaleTimeString()}
        </Text>
      )}

      {/* Share Modal */}
      <ShareCollaboratorsModal
        open={shareModalOpen}
        onClose={() => setShareModalOpen(false)}
        documentSlug={slug}
      />

      {/* Collaborative Editor */}
      <TipTapCollabEditor
        ref={editorRef}
        initialContent={doc.content}
        ydoc={ydoc}
        yjsProvider={provider}
        editable={true}
        placeholder="Start writing..."
      />
    </Box>
  );
}
